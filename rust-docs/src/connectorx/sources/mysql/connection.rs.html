7/// URL query parameters consumed here and stripped before the URL is handed to the
8/// `mysql` crate, which rejects any parameter it does not recognize.
9const SSL_PARAMS: &[&str] = &["ssl-mode", "sslmode", "ssl-ca"];
10
11/// Build the connection options for a MySQL URL, translating the `ssl-mode` (and
12/// `ssl-ca`) query parameters into a [`SslOpts`].
13///
14/// The `mysql` crate configures TLS programmatically via [`OptsBuilder::ssl_opts`] and
15/// errors on unknown URL parameters, so the SSL parameters are parsed and removed here
16/// rather than passed through to [`Opts::from_url`].
17///
18/// The crate also defaults `prefer_socket` to `true`. After a TCP handshake to a
19/// loopback address it reads the server's `socket` variable and reconnects through
20/// that Unix socket, so a forwarded port such as `127.0.0.1:9999` lands on the local
21/// server. An explicit TCP port disables that automatic preference. A URL that sets
22/// `socket` or `prefer_socket`, and a URL with no port, keeps the parsed driver value.
23pub fn build_opts(conn: &str) -> Result<OptsBuilder, MySQLSourceError> {
24    let url = Url::parse(conn)?;
25
26    let params: HashMap<String, String> = url
27        .query_pairs()
28        .map(|(k, v)| (k.to_ascii_lowercase(), v.into_owned()))
29        .collect();
30    let ssl_opts = ssl_opts_from_params(&params)?;
31
32    let mut builder = OptsBuilder::from_opts(Opts::from_url(strip_ssl_params(&url).as_str())?);
33    if explicit_tcp_port_without_socket_choice(&url) {
34        builder = builder.prefer_socket(false);
35    }
36    Ok(builder.ssl_opts(ssl_opts))
37}
38
39/// An explicit TCP port (in the authority or the `port` query) with no `socket` or
40/// `prefer_socket` parameter. Those parameters are the driver's own names, matched
41/// case-sensitively the same way [`Opts::from_url`] matches them.
42fn explicit_tcp_port_without_socket_choice(url: &Url) -> bool {
43    let mut explicit_port = url.port().is_some();
44    let mut socket_choice = false;
45    for (key, _) in url.query_pairs() {
46        match key.as_ref() {
47            "port" => explicit_port = true,
48            "socket" | "prefer_socket" => socket_choice = true,
49            _ => {}
50        }
51    }
52    explicit_port && !socket_choice
53}

55/// Translate a MySQL `ssl-mode` value into [`SslOpts`], returning `None` when TLS is not
56/// requested (no `ssl-mode`, or `ssl-mode=DISABLED`) so that behavior is unchanged for
57/// connection strings without SSL parameters.
58///
59/// `PREFERRED` is treated as `REQUIRED` (encrypt without verification): the `mysql` crate
60/// has no opportunistic mode that falls back to plaintext when the server lacks TLS.
61fn ssl_opts_from_params(
62    params: &HashMap<String, String>,
63) -> Result<Option<SslOpts>, MySQLSourceError> {
64    let mode = match params.get("ssl-mode").or_else(|| params.get("sslmode")) {
65        Some(mode) => mode,
66        None => return Ok(None),
67    };
68
69    let (accept_invalid_certs, skip_domain_validation) =
70        match mode.trim().to_ascii_uppercase().replace('-', "_").as_str() {
71            "DISABLED" => return Ok(None),
72            "PREFERRED" | "REQUIRED" => (true, true),
73            "VERIFY_CA" => (false, true),
74            "VERIFY_IDENTITY" => (false, false),
75            _ => return Err(MySQLSourceError::InvalidSslMode(mode.clone())),
76        };
77
78    let ssl_opts = SslOpts::default()
79        .with_danger_accept_invalid_certs(accept_invalid_certs)
80        .with_danger_skip_domain_validation(skip_domain_validation)
81        .with_root_cert_path(params.get("ssl-ca").map(PathBuf::from));
82    Ok(Some(ssl_opts))
83}

85fn strip_ssl_params(url: &Url) -> Url {
86    let kept: Vec<(String, String)> = url
87        .query_pairs()
88        .filter(|(k, _)| !SSL_PARAMS.contains(&k.to_ascii_lowercase().as_str()))
89        .map(|(k, v)| (k.into_owned(), v.into_owned()))
90        .collect();
91
92    let mut stripped = url.clone();
93    stripped.set_query(None);
94    for (k, v) in kept {
95        stripped.query_pairs_mut().append_pair(&k, &v);
96    }
97    stripped
98}

100#[cfg(test)]

104    fn ssl_opts(conn: &str) -> Option<SslOpts> {
105        let opts: Opts = build_opts(conn).unwrap().into();
106        opts.get_ssl_opts().cloned()

109    #[test]
110    fn no_ssl_mode_disables_tls() {
111        assert!(ssl_opts("mysql://user:pass@host:3306/db").is_none());
112        assert!(ssl_opts("mysql://user:pass@host:3306/db?ssl-mode=DISABLED").is_none());
113    }
114
115    #[test]
116    fn required_encrypts_without_verification() {
117        let opts = ssl_opts("mysql://user:pass@host:3306/db?ssl-mode=REQUIRED").unwrap();
118        assert!(opts.accept_invalid_certs());
119        assert!(opts.skip_domain_validation());
120        assert!(opts.root_cert_path().is_none());
121    }
122
123    #[test]
124    fn preferred_is_treated_as_required() {
125        assert_eq!(
126            ssl_opts("mysql://user:pass@host:3306/db?ssl-mode=PREFERRED"),
127            ssl_opts("mysql://user:pass@host:3306/db?ssl-mode=REQUIRED"),
128        );
129    }
130
131    #[test]
132    fn verify_ca_checks_chain_but_not_hostname() {
133        let opts = ssl_opts("mysql://user:pass@host:3306/db?ssl-mode=VERIFY_CA&ssl-ca=/tmp/ca.pem")
134            .unwrap();
135        assert!(!opts.accept_invalid_certs());
136        assert!(opts.skip_domain_validation());
137        assert_eq!(
138            opts.root_cert_path(),
139            Some(std::path::Path::new("/tmp/ca.pem"))
140        );
141    }

143    #[test]
144    fn verify_identity_checks_chain_and_hostname() {
145        let opts = ssl_opts("mysql://user:pass@host:3306/db?ssl-mode=VERIFY_IDENTITY").unwrap();
146        assert!(!opts.accept_invalid_certs());
147        assert!(!opts.skip_domain_validation());
148    }

150    #[test]

153        assert!(ssl_opts("mysql://user:pass@host:3306/db?sslmode=Required").is_some());

164    #[test]
165    fn ssl_params_are_stripped_so_the_url_parses_and_other_params_survive() {
166        // `ssl-ca` is not a parameter the mysql crate recognizes, so build_opts must
167        // strip it (otherwise the URL is rejected) while leaving non-SSL params intact.
168        let opts: Opts = build_opts(

177    #[test]
178    fn explicit_loopback_port_stays_on_that_port() {
179        // Automatic socket preference would drop 127.0.0.1:9999 onto the local socket.
180        let opts: Opts = build_opts("mysql://user:pass@127.0.0.1:9999/db")

196    #[test]
197    fn explicit_socket_choice_survives_on_a_url_with_a_port() {
198        let prefer_on: Opts = build_opts("mysql://user:pass@127.0.0.1:9999/db?prefer_socket=true")
199            .unwrap()
200            .into();
201        assert_eq!(prefer_on.get_tcp_port(), 9999);
202        assert!(prefer_on.get_prefer_socket());
203
204        let prefer_off: Opts =
205            build_opts("mysql://user:pass@127.0.0.1:9999/db?prefer_socket=false")
206                .unwrap()
207                .into();
208        assert_eq!(prefer_off.get_tcp_port(), 9999);
209        assert!(!prefer_off.get_prefer_socket());
210
211        let socket: Opts = build_opts("mysql://user:pass@127.0.0.1:9999/db?socket=/tmp/mysql.sock")
212            .unwrap()
213            .into();
214        let driver: Opts =
215            Opts::from_url("mysql://user:pass@127.0.0.1:9999/db?socket=/tmp/mysql.sock").unwrap();
216        assert_eq!(socket.get_tcp_port(), 9999);
217        assert_eq!(socket.get_socket(), Some("/tmp/mysql.sock"));
218        assert_eq!(socket.get_prefer_socket(), driver.get_prefer_socket());
219    }
