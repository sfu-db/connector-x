4/// Column types of the BigQuery source. The `bool` is `true` when the column is nullable.
5#[derive(Copy, Clone, Debug)]
6pub enum BigQueryTypeSystem {
7    /// BigQuery `BOOL`, read as `bool`.
8    Bool(bool),
9    /// BigQuery `BOOLEAN` (legacy name for `BOOL`), read as `bool`.
10    Boolean(bool),
11    /// BigQuery `INT64`, read as `i64`.
12    Int64(bool),
13    /// BigQuery `INTEGER` (legacy name for `INT64`), read as `i64`.
14    Integer(bool),
15    /// BigQuery `FLOAT` (legacy name for `FLOAT64`), read as `f64`.
16    Float(bool),
17    /// BigQuery `FLOAT64`, read as `f64`.
18    Float64(bool),
19    /// BigQuery `NUMERIC`, read as `f64`, so precision beyond `f64` is lost.
20    Numeric(bool),
21    /// BigQuery `BIGNUMERIC`, read as `f64`, so precision beyond `f64` is lost.
22    Bignumeric(bool),
23    /// BigQuery `STRING`, read as `String`.
24    String(bool),
25    /// BigQuery `BYTES`, read as `String` (the base64 text BigQuery returns).
26    Bytes(bool),
27    /// BigQuery `DATE`, read as `NaiveDate`.
28    Date(bool),
29    /// BigQuery `DATETIME`, read as `NaiveDateTime`.
30    Datetime(bool),
31    /// BigQuery `TIME`, read as `NaiveTime`.
32    Time(bool),
33    /// BigQuery `TIMESTAMP`, read as `DateTime<Utc>`.
34    Timestamp(bool),
35}
36
37impl_typesystem! {
38    system = BigQueryTypeSystem,
39    mappings = {
40        { Bool | Boolean => bool }
41        { Int64 | Integer  =>  i64 }
42        { Float64 | Float | Numeric | Bignumeric  =>  f64 }
43        { String | Bytes  =>  String }
44        { Date => NaiveDate }
45        { Datetime => NaiveDateTime }
46        { Time => NaiveTime }
47        { Timestamp => DateTime<Utc> }
48    }
49}

51impl<'a> From<&'a FieldType> for BigQueryTypeSystem {
52    fn from(ty: &'a FieldType) -> BigQueryTypeSystem {
53        use BigQueryTypeSystem::*;
54        match ty {
55            FieldType::Bool => Bool(true),
56            FieldType::Boolean => Boolean(true),
57            FieldType::Int64 => Int64(true),
58            FieldType::Integer => Integer(true),
59            FieldType::Float => Float(true),
60            FieldType::Float64 => Float64(true),
61            FieldType::Numeric => Numeric(true),
62            FieldType::Bignumeric => Bignumeric(true),
63            FieldType::String => String(true),
64            FieldType::Bytes => Bytes(true),
65            FieldType::Date => Date(true),
66            FieldType::Datetime => Datetime(true),
67            FieldType::Time => Time(true),
68            FieldType::Timestamp => Timestamp(true),
69            _ => unimplemented!("{}", format!("{:?}", ty)),
70        }
71    }
72}

74impl From<BigQueryTypeSystem> for FieldType {
75    fn from(ty: BigQueryTypeSystem) -> FieldType {
76        use BigQueryTypeSystem::*;
77        match ty {
78            Bool(_) => FieldType::Bool,
79            Boolean(_) => FieldType::Boolean,
80            Int64(_) => FieldType::Int64,
81            Integer(_) => FieldType::Integer,
82            Float64(_) => FieldType::Float64,
83            Float(_) => FieldType::Float,
84            Numeric(_) => FieldType::Numeric,
85            Bignumeric(_) => FieldType::Bignumeric,
86            String(_) => FieldType::String,
87            Bytes(_) => FieldType::Bytes,
88            Date(_) => FieldType::Date,
89            Datetime(_) => FieldType::Datetime,
90            Time(_) => FieldType::Time,
91            Timestamp(_) => FieldType::Timestamp,
92        }
93    }
94}

96#[cfg(test)]
97mod tests {
98    use super::BigQueryTypeSystem;
99    use gcp_bigquery_client::model::field_type::FieldType;
100
101    #[test]
102    fn maps_bigquery_field_types_in_both_directions() {
103        let fields = [
104            (FieldType::Bool, "Bool"),
105            (FieldType::Boolean, "Boolean"),
106            (FieldType::Int64, "Int64"),
107            (FieldType::Integer, "Integer"),
108            (FieldType::Float, "Float"),
109            (FieldType::Float64, "Float64"),
110            (FieldType::Numeric, "Numeric"),
111            (FieldType::Bignumeric, "Bignumeric"),
112            (FieldType::String, "String"),
113            (FieldType::Bytes, "Bytes"),
114            (FieldType::Date, "Date"),
115            (FieldType::Datetime, "Datetime"),
116            (FieldType::Time, "Time"),
117            (FieldType::Timestamp, "Timestamp"),
118        ];
119        for (field, expected) in fields {
120            assert_eq!(
121                format!("{:?}", BigQueryTypeSystem::from(&field)),
122                format!("{expected}(true)")
123            );
124            let system = BigQueryTypeSystem::from(&field);
125            assert_eq!(
126                format!("{:?}", FieldType::from(system)),
127                format!("{field:?}")
128            );
129        }
130    }
