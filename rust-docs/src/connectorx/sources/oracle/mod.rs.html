59/// Build an Oracle connector from an `oracle://` URL. Uses external authentication when no user and password are given.
60#[throws(OracleSourceError)]
61pub fn connect_oracle(conn: &Url) -> Connector {
62    let user = decode(conn.username())?.into_owned();
63    let password = decode(conn.password().unwrap_or(""))?.into_owned();
64    let host = decode(conn.host_str().unwrap_or("localhost"))?.into_owned();
65
66    let params: HashMap<String, String> = conn.query_pairs().into_owned().collect();
67
68    let conn_str = if params.get("alias").is_some_and(|v| v == "true") {
69        host.clone()
70    } else {
71        let port = conn.port().unwrap_or(1521);
72        let path = decode(conn.path())?.into_owned();
73        format!("//{}:{}{}", host, port, path)
74    };
75
76    let mut connector = oracle::Connector::new(user.as_str(), password.as_str(), conn_str.as_str());
77    if user.is_empty() && password.is_empty() {
78        debug!("No username or password provided, assuming system auth.");
79        connector.external_auth(true);
80    }
81    connector
82}

84impl OracleSource {
85    /// Create a source with a pool of up to `nconn` connections.
86    #[throws(OracleSourceError)]
87    pub fn new(conn: &str, nconn: usize) -> Self {
88        let conn = Url::parse(conn)?;
89        let connector = connect_oracle(&conn)?;
90        let manager = OracleConnectionManager::from_connector(connector);
91        let pool = r2d2::Pool::builder()
92            .max_size(nconn as u32)
93            .build(manager)?;
94
95        let params: HashMap<String, String> = conn.query_pairs().into_owned().collect();
96        let current_schema = params.get("schema").cloned();
97
98        Self {
99            pool,
100            origin_query: None,
101            queries: vec![],
102            names: vec![],
103            schema: vec![],
104            current_schema,
105        }
106    }
107    /// Get a pooled connection, switched to the `schema` from the connection string if one was given.
108    pub fn get_conn(&self) -> Result<OracleConn, OracleSourceError> {
109        let conn = self.pool.get()?;
110        if let Some(schema) = &self.current_schema {
111            conn.set_current_schema(schema)?;
112        }
113        Ok(conn)
114    }
115}

117/// Oldest Instant Client known to read native `BOOLEAN` columns correctly. On the first execution
118/// of a statement on a connection, Instant Client 23.4 can return `TRUE` for a `FALSE` value that
119/// directly follows another `FALSE`, without raising an error. Clients older than 23 report
120/// `BOOLEAN` columns as `NUMBER`, so they never reach this check.
121const MIN_BOOLEAN_CLIENT_VERSION: Version = Version::new(23, 5, 0, 0, 0);
122
123fn client_reads_boolean_correctly(client: &Version) -> bool {
124    *client >= MIN_BOOLEAN_CLIENT_VERSION
125}
126
127/// Refuses to read a native `BOOLEAN` column through an Instant Client that may return wrong
128/// values for it.
129#[throws(OracleSourceError)]
130fn check_boolean_client(names: &[String], schema: &[OracleTypeSystem]) {
131    if let Some(i) = schema
132        .iter()
133        .position(|ty| matches!(ty, OracleTypeSystem::Boolean(_)))
134    {
135        let client = Version::client()?;
136        if !client_reads_boolean_correctly(&client) {
137            throw!(OracleSourceError::UnreliableBooleanClient {
138                column: names[i].clone(),
139                client: client.to_string(),
140            });
141        }
142    }
143}

145impl Source for OracleSource
146where
147    OracleSourcePartition:
148        SourcePartition<TypeSystem = OracleTypeSystem, Error = OracleSourceError>,
149{
150    const DATA_ORDERS: &'static [DataOrder] = &[DataOrder::RowMajor];
151    type Partition = OracleSourcePartition;
152    type TypeSystem = OracleTypeSystem;
153    type Error = OracleSourceError;
154
155    #[throws(OracleSourceError)]
156    fn set_data_order(&mut self, data_order: DataOrder) {
157        if !matches!(data_order, DataOrder::RowMajor) {
158            throw!(ConnectorXError::UnsupportedDataOrder(data_order));
159        }
160    }
161
162    fn set_queries<Q: ToString>(&mut self, queries: &[CXQuery<Q>]) {
163        self.queries = queries.iter().map(|q| q.map(Q::to_string)).collect();
164    }
165
166    fn set_origin_query(&mut self, query: Option<String>) {
167        self.origin_query = query;
168    }
169
170    #[throws(OracleSourceError)]
171    fn fetch_metadata(&mut self) {
172        assert!(!self.queries.is_empty());
173
174        let conn = self.get_conn()?;
175        for (i, query) in self.queries.iter().enumerate() {
176            // assuming all the partition queries yield same schema
177            // without rownum = 1, derived type might be wrong
178            // example: select avg(test_int), test_char from test_table group by test_char
179            // -> (NumInt, Char) instead of (NumtFloat, Char)
180            match conn.query(limit0_query_oracle(query)?.as_str(), &[]) {
181                Ok(rows) => {
182                    let (names, types): (Vec<String>, Vec<OracleTypeSystem>) = rows
183                        .column_info()
184                        .iter()
185                        .map(|col| {
186                            (
187                                col.name().to_string(),
188                                OracleTypeSystem::from(col.oracle_type()),
189                            )
190                        })
191                        .unzip();
192                    check_boolean_client(&names, &types)?;
193                    self.names = names;
194                    self.schema = types;
195                    return;
196                }
197                Err(e) if i == self.queries.len() - 1 => {
198                    // tried the last query but still get an error
199                    debug!("cannot get metadata for '{}': {}", query, e);
200                    throw!(e);
201                }
202                Err(_) => {}
203            }
204        }
205        // tried all queries but all get empty result set
206        let iter = conn.query(self.queries[0].as_str(), &[])?;
207        let (names, types) = iter
208            .column_info()
209            .iter()
210            .map(|col| (col.name().to_string(), OracleTypeSystem::VarChar(false)))
211            .unzip();
212        self.names = names;
213        self.schema = types;
214    }
215
216    #[throws(OracleSourceError)]
217    fn result_rows(&mut self) -> Option<usize> {
218        match &self.origin_query {
219            Some(q) => {
220                let cxq = CXQuery::Naked(q.clone());
221                let conn = self.get_conn()?;
222
223                let nrows = conn
224                    .query_row_as::<usize>(count_query(&cxq, &OracleDialect {})?.as_str(), &[])?;
225                Some(nrows)
226            }
227            None => None,
228        }
229    }
230
231    fn names(&self) -> Vec<String> {
232        self.names.clone()
233    }
234
235    fn schema(&self) -> Vec<Self::TypeSystem> {
236        self.schema.clone()
237    }
238
239    #[throws(OracleSourceError)]
240    fn partition(self) -> Vec<Self::Partition> {
241        let mut ret = vec![];
242        for query in &self.queries {
243            let conn = self.get_conn()?;
244            ret.push(OracleSourcePartition::new(conn, query, &self.schema));
245        }
246        ret
247    }
248}

303/// Row parser for an Oracle partition.
304pub struct OracleTextSourceParser<'a> {
305    rows: OwningHandle<Statement, DummyBox<ResultSet<'a, Row>>>,
306    rowbuf: Vec<Row>,
307    ncols: usize,
308    current_col: usize,
309    current_row: usize,
310    is_finished: bool,
311}
312
313impl<'a> OracleTextSourceParser<'a> {
314    /// Run `query` on `conn` and prepare to parse its rows.
315    #[throws(OracleSourceError)]
316    pub fn new(conn: &'a OracleConn, query: &str, schema: &[OracleTypeSystem]) -> Self {
317        let stmt = conn
318            .statement(query)
319            .prefetch_rows(ORACLE_ARRAY_SIZE)
320            .fetch_array_size(ORACLE_ARRAY_SIZE)
321            .build()?;
322        let rows: OwningHandle<Statement, DummyBox<ResultSet<'a, Row>>> =
323            OwningHandle::new_with_fn(Box::new(stmt), |stmt: *const Statement| unsafe {
324                DummyBox((*(stmt as *mut Statement)).query(&[]).unwrap())
325            });
326
327        Self {
328            rows,
329            rowbuf: Vec::with_capacity(DB_BUFFER_SIZE),
330            ncols: schema.len(),
331            current_row: 0,
332            current_col: 0,
333            is_finished: false,
334        }
335    }
336
337    #[throws(OracleSourceError)]
338    fn next_loc(&mut self) -> (usize, usize) {
339        let ret = (self.current_row, self.current_col);
340        self.current_row += (self.current_col + 1) / self.ncols;
341        self.current_col = (self.current_col + 1) % self.ncols;
342        ret
343    }
344}

346impl<'a> PartitionParser<'a> for OracleTextSourceParser<'a> {
347    type TypeSystem = OracleTypeSystem;
348    type Error = OracleSourceError;
349
350    #[throws(OracleSourceError)]
351    fn fetch_next(&mut self) -> (usize, bool) {
352        assert!(self.current_col == 0);
353        let remaining_rows = self.rowbuf.len() - self.current_row;
354        if remaining_rows > 0 {
355            return (remaining_rows, self.is_finished);
356        } else if self.is_finished {
357            return (0, self.is_finished);
358        }
359
360        if !self.rowbuf.is_empty() {
361            self.rowbuf.drain(..);
362        }
363        for _ in 0..DB_BUFFER_SIZE {
364            if let Some(item) = (*self.rows).next() {
365                self.rowbuf.push(item?);
366            } else {
367                self.is_finished = true;
368                break;
369            }
370        }
371        self.current_row = 0;
372        self.current_col = 0;
373        (self.rowbuf.len(), self.is_finished)
374    }
375}

407// `TIMESTAMP WITH TIME ZONE`: rust-oracle's `FromSql for DateTime<Utc>` reads the wall-clock time as if it were UTC
408// and drops the offset (`11:00:00 -08:00` would become `11:00:00 UTC`). Read it with its offset, then convert.
409fn timestamp_tz_to_utc(value: DateTime<FixedOffset>) -> DateTime<Utc> {
410    value.with_timezone(&Utc)
411}

413impl<'r, 'a> Produce<'r, DateTime<Utc>> for OracleTextSourceParser<'a> {
414    type Error = OracleSourceError;
415
416    #[throws(OracleSourceError)]
417    fn produce(&'r mut self) -> DateTime<Utc> {
418        let (ridx, cidx) = self.next_loc()?;
419        let value: DateTime<FixedOffset> = self.rowbuf[ridx].get(cidx)?;
420        timestamp_tz_to_utc(value)
421    }
422}
