1//! Source implementation for SQL Server.
2//!
3//! Two backend implementations exist, selected at compile time by mutually
4//! exclusive Cargo features:
5//! - `src_mssql_tiberius` (default): `tiberius` + `bb8-tiberius`.
6//! - `src_mssql_tds`: `mssql-tds`.
7//!
8//! Both compile against the same [`typesystem::MsSQLTypeSystem`] and
9//! [`errors::MsSQLSourceError`], so [`crate::transports::mssql_arrow`] and
10//! everything above the source layer is unaffected by which backend is
11//! active. See sfu-db/connector-x#942 for the migration plan.
12
13#[cfg(all(feature = "src_mssql_tiberius", feature = "src_mssql_tds"))]
14compile_error!(
15    "features `src_mssql_tiberius` and `src_mssql_tds` are mutually exclusive; pick one MSSQL backend"
16);
17
18#[cfg(all(not(feature = "src_mssql_tiberius"), not(feature = "src_mssql_tds")))]
19compile_error!("MSSQL source requires either `src_mssql_tiberius` or `src_mssql_tds`");
