1//! `mssql-tds`-backed implementation of the MsSQL source.
2//!
3//! Added in Phase 2 of the tiberius -> mssql-tds migration (see
4//! sfu-db/connector-x#942). Compiled only when the `src_mssql_tds` feature
5//! is active. This implementation targets the same
6//! [`super::typesystem::MsSQLTypeSystem`] and the same set of native Rust
7//! produce types as [`super::tiberius_impl`], so
8//! [`crate::transports::mssql_arrow::MsSQLArrowTransport`] works unchanged
9//! against either backend.
10//!
11//! Automatic partition-range discovery supports integer and floating-point
12//! columns, matching Tiberius's NULL and float-to-integer conversions.
13//! Metadata, counts and partition readers share a per-source bounded bb8 pool.
14//! Partitions acquire connections only when executing, so their number may
15//! exceed the pool size.
16//!
17//! Known, intentional differences from the Tiberius path (tracked as
18//! documented gaps, not bugs):
19//! - `trust_server_certificate_ca` (validate the server cert against a
20//!   specific CA bundle without disabling hostname/chain checks) is
21//!   rejected outright: `mssql-tds`'s `EncryptionOptions::server_certificate`
22//!   only supports exact-match certificate pinning, not CA validation.
23//! - The default TLS posture differs: Tiberius defaults to
24//!   `EncryptionLevel::NotSupported` (advertise TLS as unsupported) when
25//!   `encrypt` is unset (prelogin 0x02). `mssql-tds` has no equivalent public
26//!   setting: unset and `encrypt=false` both map to `PreferOff` (0x00).
27//!   This permits login-only TLS, or full-session TLS if the server requires it.
28//!   The driver's login-only TLS unconditionally skips certificate-chain validation,
29//!   even when `trust_server_certificate=false`. Use `encrypt=true` for required
30//!   full-session TLS (0x03), with certificate-chain validation enabled unless
31//!   `trust_server_certificate=true` is explicitly requested.
32

77/// Sets the wrapper's package version and runtime for the TDS User-Agent, leaving LOGIN7 unchanged.
78/// Language bindings should call this before creating connections. Reinitializing
79/// with the same values is allowed; different values are rejected.
80pub fn set_user_agent_info(version: String, runtime: String) -> Result<(), &'static str> {
81    if version.is_empty() {
82        return Err("MSSQL User-Agent version must not be empty");
83    }
84    if runtime.is_empty() {
85        return Err("MSSQL User-Agent runtime must not be empty");
86    }
87    let info = UserAgentInfo { version, runtime };
88    if USER_AGENT_INFO.get_or_init(|| info.clone()) != &info {
89        return Err("MSSQL User-Agent is already initialized with different values");
90    }
91    Ok(())
92}

94/// Builds the `mssql-tds` datasource string and [`ClientContext`] from a
95/// ConnectorX MSSQL connection URL. Mirrors
96/// [`super::tiberius_impl::mssql_config`]'s parsing, but expressed against
97/// `mssql-tds`'s config surface.
98#[throws(MsSQLSourceError)]
99fn build_client_context(url: &Url) -> (String, ClientContext) {
100    let host = decode(url.host_str().unwrap_or("localhost"))?.into_owned();
101    let port = url.port().unwrap_or(1433);
102    let hosts: Vec<&str> = host.split('\\').collect();
103    let datasource = match hosts.len() {
104        1 => format!("tcp:{},{}", hosts[0], port),
105        // SQL Server named instance: `server\instance`. Let mssql-tds's own
106        // SSRP resolution figure out the port, same as Tiberius does today.
107        2 => format!("{}\\{}", hosts[0], hosts[1]),
108        _ => throw!(anyhow!("MsSQL hostname parse error: {}", host)),
109    };
110
111    let mut context = ClientContext::with_data_source(&datasource);
112    // The User-Agent feature carries a separate driver name from LOGIN7.
113    context
114        .user_agent
115        .set_library_name("connectorx".to_string());
116    if let Some(info) = USER_AGENT_INFO.get() {
117        context.user_agent.set_driver_version(info.version.clone());
118        context.user_agent.set_runtime(info.runtime.clone());
119    }
120    context.application_name = "ConnectorX".to_string();
121    context.database = decode(&url.path()[1..])?.into_owned();
122
123    let params: HashMap<String, String> = url.query_pairs().into_owned().collect();
124
125    match params.get("trusted_connection") {
126        Some(v) if v == "true" && cfg!(any(windows, feature = "integrated-auth-gssapi")) => {
127            debug!("mssql-tds auth through integrated (SSPI) authentication");
128            context.tds_authentication_method = TdsAuthenticationMethod::SSPI;
129        }
130        _ => {
131            debug!("mssql-tds auth through sqlserver authentication");
132            context.tds_authentication_method = TdsAuthenticationMethod::Password;
133            context.user_name = decode(url.username())?.into_owned();
134            context.password = decode(url.password().unwrap_or(""))?.into_owned();
135        }
136    }
137
138    if let Some(v) = params.get("trust_server_certificate_ca") {
139        // See the module-level doc comment: mssql-tds has no CA-validation
140        // equivalent yet, only exact-match certificate pinning. Reject
141        // rather than silently downgrade TLS trust.
142        throw!(anyhow!(
143            "trust_server_certificate_ca={} is not supported with the src_mssql_tds backend: \
144             mssql-tds only supports exact-match certificate pinning, not CA validation",
145            v
146        ));
147    }
148
149    let mut encryption = EncryptionOptions::new();
150    encryption.mode = EncryptionSetting::PreferOff;
151    match params.get("encrypt") {
152        Some(v) if v.to_lowercase() == "true" => encryption.mode = EncryptionSetting::Required,
153        Some(v) if v.to_lowercase() == "false" => encryption.mode = EncryptionSetting::PreferOff,
154        _ => {}
155    }
156    if let Some(v) = params.get("trust_server_certificate") {
157        if v.to_lowercase() == "true" {
158            encryption.trust_server_certificate = true;
159        }
160    }
161    context.encryption_options = encryption;
162
163    if let Some(appname) = params.get("appname") {
164        context.application_name = decode(appname)?.into_owned();
165    }
166
167    (datasource, context)
168}

308struct ConnectionManager {
309    url: Url,
310}
311
312impl ManageConnection for ConnectionManager {
313    type Connection = TdsClient;
314    type Error = MsSQLSourceError;
315
316    async fn connect(&self) -> Result<TdsClient, Self::Error> {
317        let (datasource, context) = build_client_context(&self.url)?;
318        Ok(TdsConnectionProvider::new()
319            .create_client(context, &datasource, None)
320            .await?)
321    }
322
323    async fn is_valid(&self, client: &mut TdsClient) -> Result<(), Self::Error> {
324        if client.is_connection_dead() {
325            return Err(anyhow!("MsSQL pooled connection is closed").into());
326        }
327        // Reset session settings and transactions before the next borrower's query.
328        client.prepare_reset_connection(false);
329        Ok(())
330    }
331
332    fn has_broken(&self, client: &mut TdsClient) -> bool {
333        client.is_connection_dead() || client.has_open_batch()
334    }
335}
336
337#[throws(MsSQLSourceError)]
338fn connection_pool(rt: &Runtime, url: &Url, nconn: usize) -> Pool<ConnectionManager> {
339    let max_size = u32::try_from(nconn)
340        .ok()
341        .filter(|&size| size > 0)
342        .ok_or_else(|| anyhow!("MsSQL pool size must be between 1 and {}", u32::MAX))?;
343    // Fail on unsupported URL options even before the first checkout.
344    build_client_context(url)?;
345    rt.block_on(
346        Pool::builder()
347            .max_size(max_size)
348            .test_on_check_out(true)
349            .build(ConnectionManager { url: url.clone() }),
350    )?
351}

353/// Owns the lease and keeps its runtime alive through cleanup and pool return.
354struct Connection {
355    rt: Arc<Runtime>,
356    lease: Option<PooledConnection<'static, ConnectionManager>>,
357}
358
359impl Connection {
360    #[throws(MsSQLSourceError)]
361    fn get(rt: Arc<Runtime>, pool: &Pool<ConnectionManager>) -> Self {
362        let lease = rt.block_on(pool.get_owned()).map_err(|err| match err {
363            bb8::RunError::User(err) => err,
364            bb8::RunError::TimedOut => anyhow!("Timed out waiting for an MsSQL connection").into(),
365        })?;
366        Self {
367            rt,
368            lease: Some(lease),
369        }
370    }
371
372    #[throws(MsSQLSourceError)]
373    fn finish(&mut self) {
374        let client = self.lease.as_mut().unwrap();
375        if let Err(err) = self.rt.block_on(client.close_query()) {
376            client.mark_connection_dead();
377            throw!(err);
378        }
379    }
380}
381
382impl Deref for Connection {
383    type Target = TdsClient;
384
385    fn deref(&self) -> &TdsClient {
386        self.lease.as_ref().unwrap()
387    }
388}
389
390impl DerefMut for Connection {
391    fn deref_mut(&mut self) -> &mut TdsClient {
392        self.lease.as_mut().unwrap()
393    }
394}
395
396impl Drop for Connection {
397    fn drop(&mut self) {
398        let mut client = self.lease.take().unwrap();
399        if !client.is_connection_dead() && client.has_open_batch() {
400            // Early parser drops must not return unread protocol data to the pool.
401            // Bound destructor work; a timed-out drain is discarded, never reused.
402            match self.rt.block_on(async {
403                tokio::time::timeout(Duration::from_secs(5), client.close_query()).await
404            }) {
405                Ok(Ok(())) => {}
406                Ok(Err(err)) => {
407                    warn!("Discarding MsSQL connection after cleanup failure: {}", err);
408                    client.mark_connection_dead();
409                }
410                Err(_) => {
411                    warn!("Discarding MsSQL connection after cleanup timeout");
412                    client.mark_connection_dead();
413                }
414            }
415        }
416        // bb8 may spawn replacement connections when returning a broken lease.
417        let _guard = self.rt.enter();
418        drop(client);
419    }
420}

423#[throws(MsSQLSourceError)]
424fn first_row(rt: &Runtime, client: &mut TdsClient, sql: String) -> Option<Vec<ColumnValues>> {

429#[throws(MsSQLSourceError)]
430pub(crate) fn tds_get_partition_range(url: &Url, query: &str, col: &str) -> (i64, i64) {
431    let range_query = get_partition_range_query(query, col, &MsSqlDialect {})?;
432    let rt = Runtime::new().map_err(anyhow::Error::from)?;
433    let mut client = open_connection(&rt, url)?;
434    rt.block_on(client.execute(range_query, ()))?;
435    let types: Vec<_> = client
436        .get_metadata()
437        .iter()
438        .map(|column| column.data_type)
439        .collect();
440    let row = rt
441        .block_on(client.next_row())?
442        .ok_or_else(|| anyhow!("MsSQL partition range query returned no row"))?;
443    if row.len() != 2 || types.len() != 2 {
444        throw!(anyhow!(
445            "MsSQL partition range query must return two columns"
446        ));
447    }
448    (
449        partition_range_value(&row[0], types[0])?,
450        partition_range_value(&row[1], types[1])?,
451    )
452}
453
454#[throws(MsSQLSourceError)]
455fn partition_range_value(value: &ColumnValues, ty: TdsDataType) -> i64 {
456    if !matches!(
457        ty,
458        TdsDataType::Int1
459            | TdsDataType::Int2
460            | TdsDataType::Int4
461            | TdsDataType::Int8
462            | TdsDataType::IntN
463            | TdsDataType::Flt4
464            | TdsDataType::Flt8
465            | TdsDataType::FltN
466    ) {
467        throw!(anyhow!(
468            "Partition can only be done on int or float columns"
469        ));
470    }
471    // Match Tiberius: NULL aggregates become zero and floats truncate to i64.
472    match value {
473        ColumnValues::Null => 0,
474        ColumnValues::TinyInt(n) => i64::from(*n),
475        ColumnValues::SmallInt(n) => i64::from(*n),
476        ColumnValues::Int(n) => i64::from(*n),
477        ColumnValues::BigInt(n) => *n,
478        ColumnValues::Real(n) => *n as i64,
479        ColumnValues::Float(n) => *n as i64,
480        other => throw!(anyhow!(
481            "Unexpected MsSQL partition range value: {:?}",
482            other
483        )),
484    }
485}

557
558impl Source for MsSQLSource
559where
560    MsSQLSourcePartition: SourcePartition<TypeSystem = MsSQLTypeSystem, Error = MsSQLSourceError>,
561{
562    const DATA_ORDERS: &'static [DataOrder] = &[DataOrder::RowMajor];
563    type Partition = MsSQLSourcePartition;
564    type TypeSystem = MsSQLTypeSystem;
565    type Error = MsSQLSourceError;
566
567    #[throws(MsSQLSourceError)]
568    fn set_data_order(&mut self, data_order: DataOrder) {
569        if !matches!(data_order, DataOrder::RowMajor) {
570            throw!(ConnectorXError::UnsupportedDataOrder(data_order));
571        }
572    }
573
574    fn set_queries<Q: ToString>(&mut self, queries: &[CXQuery<Q>]) {
575        self.queries = queries.iter().map(|q| q.map(Q::to_string)).collect();
576    }
577
578    fn set_origin_query(&mut self, query: Option<String>) {
579        self.origin_query = query;
580    }
581
582    #[throws(MsSQLSourceError)]
583    fn fetch_metadata(&mut self) {
584        assert!(!self.queries.is_empty());
585
586        let first_query = self.queries[0].clone();
587        let mut client = Connection::get(self.rt.clone(), &self.pool)?;
588        self.rt
589            .block_on(client.execute(first_query.as_str().to_string(), ()))?;
590
591        let metadata = client.get_metadata();
592        if metadata.is_empty() {
593            throw!(anyhow!(
594                "MsSQL returned no columns for query: {}",
595                first_query
596            ));
597        }
598
599        let (names, types) = metadata
600            .iter()
601            .map(|col| (col.column_name.clone(), MsSQLTypeSystem::from(col)))
602            .unzip();
603        client.finish()?;
604
605        self.names = names;
606        self.schema = types;
607    }
608
609    #[throws(MsSQLSourceError)]
610    fn result_rows(&mut self) -> Option<usize> {
611        match &self.origin_query {
612            Some(q) => {
613                let cxq = CXQuery::Naked(q.clone());
614                let cquery = count_query(&cxq, &MsSqlDialect {})?;
615                let mut client = Connection::get(self.rt.clone(), &self.pool)?;
616                let row = first_row(&self.rt, &mut client, cquery.as_str().to_string())?
617                    .ok_or_else(|| anyhow!("MsSQL failed to get the count of query: {}", q))?;
618                client.finish()?;
619                Some(row_count_value(&row, q)?)
620            }
621            None => None,
622        }
623    }
624
625    fn names(&self) -> Vec<String> {
626        self.names.clone()
627    }
628
629    fn schema(&self) -> Vec<Self::TypeSystem> {
630        self.schema.clone()
631    }
632
633    #[throws(MsSQLSourceError)]
634    fn partition(self) -> Vec<Self::Partition> {
635        let mut ret = vec![];
636        for query in self.queries {
637            let mut partition = MsSQLSourcePartition::new(
638                self.rt.clone(),
639                self.conn_url.clone(),
640                &query,
641                &self.schema,
642            );
643            partition.pool = Some(self.pool.clone());
644            ret.push(partition);
645        }
646        ret
647    }
648}

650/// Reads the first column of a `COUNT(*)` result row. SQL Server always
651/// returns `COUNT(*)` as `int`.
652#[throws(MsSQLSourceError)]
653fn row_count_value(row: &[ColumnValues], query: &str) -> usize {
654    match row.first() {
655        Some(ColumnValues::Int(n)) => *n as usize,
656        Some(ColumnValues::BigInt(n)) => *n as usize,
657        other => throw!(anyhow!(
658            "MsSQL count query for '{}' returned unexpected value: {:?}",
659            query,
660            other
661        )),
662    }

736/// One decoded, native-Rust representation of a single `ColumnValues` cell.
737/// Converting eagerly (once, when the row arrives) rather than on every
738/// `Produce` call keeps the per-cell conversion logic (decimal, date/time,
739/// money math) in one place and lets `&str`/`&[u8]` `Produce` impls borrow
740/// directly from this buffer.
741#[derive(Debug)]
742enum TdsCell {
743    Null,
744    U8(u8),
745    I16(i16),
746    I32(i32),
747    I64(i64),
748    F32(f32),
749    F64(f64),
750    Bool(bool),
751    Str(String),
752    Bytes(Vec<u8>),
753    Uuid(Uuid),
754    Decimal(Decimal),
755    NaiveDate(NaiveDate),
756    NaiveTime(NaiveTime),
757    NaiveDateTime(NaiveDateTime),
758    DateTimeUtc(DateTime<Utc>),
759}
760
761pub struct MsSQLSourceParser {
762    rt: Arc<Runtime>,
763    client: Connection,
764    rowbuf: Vec<Vec<TdsCell>>,
765    ncols: usize,
766    current_col: usize,
767    current_row: usize,
768    is_finished: bool,
769}
770
771impl MsSQLSourceParser {
772    fn new(rt: Arc<Runtime>, client: Connection, ncols: usize) -> Self {
773        Self {
774            rt,
775            client,
776            rowbuf: Vec::with_capacity(DB_BUFFER_SIZE),
777            ncols,
778            current_row: 0,
779            current_col: 0,
780            is_finished: false,
781        }
782    }
783
784    #[throws(MsSQLSourceError)]
785    fn next_loc(&mut self) -> (usize, usize) {
786        let ret = (self.current_row, self.current_col);
787        self.current_row += (self.current_col + 1) / self.ncols;
788        self.current_col = (self.current_col + 1) % self.ncols;
789        ret
790    }
791
792    /// Takes ownership of a cell out of the row buffer, leaving `Null`
793    /// behind. Safe because each cell is produced exactly once per row (see
794    /// `next_loc`).
795    fn take_cell(&mut self, ridx: usize, cidx: usize) -> TdsCell {
796        std::mem::replace(&mut self.rowbuf[ridx][cidx], TdsCell::Null)
797    }
798}

800impl<'a> PartitionParser<'a> for MsSQLSourceParser {
801    type TypeSystem = MsSQLTypeSystem;
802    type Error = MsSQLSourceError;
803
804    #[throws(MsSQLSourceError)]
805    fn fetch_next(&mut self) -> (usize, bool) {
806        assert!(self.current_col == 0);
807        let remaining_rows = self.rowbuf.len() - self.current_row;
808        if remaining_rows > 0 {
809            return (remaining_rows, self.is_finished);
810        } else if self.is_finished {
811            return (0, self.is_finished);
812        }
813
814        self.rowbuf.clear();
815
816        for _ in 0..DB_BUFFER_SIZE {
817            match self.rt.block_on(self.client.next_row())? {
818                Some(row) => {
819                    let mut cells = Vec::with_capacity(row.len());
820                    for value in row {
821                        cells.push(column_value_to_cell(value)?);
822                    }
823                    self.rowbuf.push(cells);
824                }
825                None => {
826                    self.client.finish()?;
827                    self.is_finished = true;
828                    break;
829                }
830            }
831        }
832        self.current_row = 0;
833        self.current_col = 0;
834        (self.rowbuf.len(), self.is_finished)
835    }
836}

838#[throws(MsSQLSourceError)]
839fn column_value_to_cell(value: ColumnValues) -> TdsCell {
840    match value {
841        ColumnValues::Null => TdsCell::Null,
842        ColumnValues::TinyInt(n) => TdsCell::U8(n),
843        ColumnValues::SmallInt(n) => TdsCell::I16(n),
844        ColumnValues::Int(n) => TdsCell::I32(n),
845        ColumnValues::BigInt(n) => TdsCell::I64(n),
846        ColumnValues::Real(f) => TdsCell::F32(f),
847        ColumnValues::Float(f) => TdsCell::F64(f),
848        ColumnValues::Bit(b) => TdsCell::Bool(b),
849        ColumnValues::String(s) => TdsCell::Str(s.to_utf8_string()),
850        ColumnValues::Bytes(b) => TdsCell::Bytes(b),
851        ColumnValues::Uuid(u) => TdsCell::Uuid(Uuid::from_bytes(*u.as_bytes())),
852        ColumnValues::Decimal(parts) | ColumnValues::Numeric(parts) => {
853            TdsCell::Decimal(decimal_parts_to_decimal(&parts)?)
854        }
855        ColumnValues::Date(d) => TdsCell::NaiveDate(sql_date_to_naive_date(&d)?),
856        ColumnValues::Time(t) => TdsCell::NaiveTime(sql_time_to_naive_time(&t)),
857        ColumnValues::DateTime(dt) => TdsCell::NaiveDateTime(sql_datetime_to_naive_datetime(&dt)?),
858        ColumnValues::DateTime2(dt2) => {
859            TdsCell::NaiveDateTime(sql_datetime2_to_naive_datetime(&dt2)?)
860        }
861        ColumnValues::SmallDateTime(sdt) => {
862            TdsCell::NaiveDateTime(sql_smalldatetime_to_naive_datetime(&sdt))
863        }
864        ColumnValues::DateTimeOffset(dto) => {
865            TdsCell::DateTimeUtc(sql_datetimeoffset_to_datetime_utc(&dto)?)
866        }
867        ColumnValues::Money(m) => TdsCell::F64(sql_money_to_f64(&m)),
868        // Always widened to f64, never rounded through f32: Tiberius's own
869        // money decoder (`money::decode`) computes the value as f64 in every
870        // case (`src.read_i32_le() as f64 / 1e4` for the 4-byte form too),
871        // regardless of whether the static wire type is the fixed `Money4`
872        // or the nullable `MONEYN`. Only the destination typesystem variant
873        // (`typesystem.rs`) differs by nullability, and for the nullable
874        // case it's `Money`/f64 (see there) - so keep full f64 precision
875        // here and let `Produce<f32>` narrow it only if a genuinely
876        // non-nullable smallmoney column ever needs it.
877        ColumnValues::SmallMoney(sm) => TdsCell::F64((sm.int_val as f64) / 10_000.0),
878        other => throw!(anyhow!(
879            "MsSQL: unsupported value for src_mssql_tds backend: {:?}",
880            other
881        )),
882    }
883}

885/// Reassembles a `DecimalParts` value into a `rust_decimal::Decimal` using its
886/// public `magnitude()` accessor (already scaled by `10^scale`, unsigned).
887///
888/// `rust_decimal::Decimal` caps out around a 96-bit mantissa (scale 0-28);
889/// this mirrors a limitation ConnectorX's existing Tiberius path already
890/// has today (see `utils::decimal_to_i128`), not a new one introduced here.
891#[throws(MsSQLSourceError)]
892fn decimal_parts_to_decimal(parts: &DecimalParts) -> Decimal {
893    let magnitude = parts.magnitude();
894    let mantissa = i128::try_from(magnitude)
895        .map_err(|_| anyhow!("MsSQL decimal/numeric value overflows 128 bits"))?;
896    let mantissa = if parts.is_positive {
897        mantissa
898    } else {
899        -mantissa
900    };
901    Decimal::try_from_i128_with_scale(mantissa, parts.scale as u32)
902        .map_err(|e| anyhow!("MsSQL decimal/numeric value out of range: {}", e))?
903}

905/// `SqlDate::get_days()` is days since `0001-01-01`, matching chrono's
906/// proleptic Gregorian calendar directly (no epoch conversion needed).
907#[throws(MsSQLSourceError)]
908fn sql_date_to_naive_date(d: &SqlDate) -> NaiveDate {
909    NaiveDate::from_ymd_opt(1, 1, 1)
910        .unwrap()
911        .checked_add_signed(chrono::Duration::days(d.get_days() as i64))
912        .ok_or_else(|| {
913            anyhow!(
914                "MsSQL date out of range: {} days since 0001-01-01",
915                d.get_days()
916            )
917        })?
918}

920/// Despite its name, `SqlTime::time_nanoseconds` is actually the value in
921/// 100-nanosecond ticks since midnight, not raw nanoseconds - confirmed by
922/// reading `mssql-tds`'s own `scale_time_value()` decoder helper, which
923/// documents "Expands a time value at TDS scale 0..=7 into 100-nanosecond
924/// ticks." The field's doc comment ("Nanoseconds since midnight") does not
925/// match the actual unit; multiply by 100 to get true nanoseconds.
926fn sql_time_to_naive_time(t: &SqlTime) -> NaiveTime {
927    let ns = t.time_nanoseconds * 100;
928    let secs = (ns / 1_000_000_000) as u32;
929    let nanos = (ns % 1_000_000_000) as u32;
930    NaiveTime::from_num_seconds_from_midnight_opt(secs, nanos).unwrap_or_default()
931}

933#[throws(MsSQLSourceError)]
934fn sql_datetime2_to_naive_datetime(dt2: &SqlDateTime2) -> NaiveDateTime {
935    let date = NaiveDate::from_ymd_opt(1, 1, 1)
936        .unwrap()
937        .checked_add_signed(chrono::Duration::days(dt2.days as i64))
938        .ok_or_else(|| {
939            anyhow!(
940                "MsSQL datetime2 out of range: {} days since 0001-01-01",
941                dt2.days
942            )
943        })?;
944    date.and_time(sql_time_to_naive_time(&dt2.time))
945}

947fn sql_smalldatetime_to_naive_datetime(sdt: &SqlSmallDateTime) -> NaiveDateTime {
948    // Whole minutes since midnight - no rounding needed.
949    let date = NaiveDate::from_ymd_opt(1900, 1, 1)
950        .unwrap()
951        .checked_add_signed(chrono::Duration::days(sdt.days as i64))
952        .expect("smalldatetime day count is always in range (u16 days since 1900-01-01)");
953    let time = NaiveTime::from_hms_opt((sdt.time / 60) as u32, (sdt.time % 60) as u32, 0)
954        .unwrap_or_default();
955    date.and_time(time)
956}

958/// The legacy `datetime` type stores 1/300-second ticks since midnight,
959/// rounded by SQL Server to the nearest tick when the value was written.
960/// Converting back to milliseconds requires the classic
961/// `round(ticks * 10 / 3)` algorithm; rounding a tick count near midnight
962/// can round up into the next day, which must be reflected in the date.
963#[throws(MsSQLSourceError)]
964fn sql_datetime_to_naive_datetime(dt: &SqlDateTime) -> NaiveDateTime {
965    let total_ms = (dt.time as i64 * 10 + 1) / 3;
966    let (extra_days, ms_in_day) = if total_ms >= 86_400_000 {
967        (1i64, total_ms - 86_400_000)
968    } else {
969        (0, total_ms)
970    };
971
972    let time = NaiveTime::from_num_seconds_from_midnight_opt(
973        (ms_in_day / 1000) as u32,
974        ((ms_in_day % 1000) * 1_000_000) as u32,
975    )
976    .ok_or_else(|| anyhow!("MsSQL datetime: invalid tick count {}", dt.time))?;
977
978    let date = NaiveDate::from_ymd_opt(1900, 1, 1)
979        .unwrap()
980        .checked_add_signed(chrono::Duration::days(dt.days as i64 + extra_days))
981        .ok_or_else(|| {
982            anyhow!(
983                "MsSQL datetime out of range: {} days since 1900-01-01",
984                dt.days
985            )
986        })?;
987
988    date.and_time(time)
989}

991#[throws(MsSQLSourceError)]
992fn sql_datetimeoffset_to_datetime_utc(dto: &SqlDateTimeOffset) -> DateTime<Utc> {
993    // Confirmed against a live container (not the wire-format assumption
994    // originally documented here): the datetime2 component of a
995    // DATETIMEOFFSET value is transmitted as the original *local* wall-clock
996    // time, with `offset` (minutes, e.g. 60 for +01:00) recording how far
997    // ahead of UTC it is. UTC = local - offset.
998    let naive_local = sql_datetime2_to_naive_datetime(&dto.datetime2)?;
999    let naive_utc = naive_local - chrono::Duration::minutes(dto.offset as i64);
1000    DateTime::<Utc>::from_naive_utc_and_offset(naive_utc, Utc)
1001}

1003/// Reassembles the mixed-endian 64-bit integer from `SqlMoney`'s two 32-bit
1004/// halves and scales it down (money is stored as the value * 10^4).
1005fn sql_money_to_f64(m: &mssql_tds::datatypes::column_values::SqlMoney) -> f64 {
1006    let lsb_in_i64 = (m.lsb_part as i64) & 0x0000_0000_FFFF_FFFF;
1007    let money_val = lsb_in_i64 | ((m.msb_part as i64) << 32);
1008    (money_val as f64) / 10_000.0
1009}
