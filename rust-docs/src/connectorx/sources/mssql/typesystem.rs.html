7// https://docs.microsoft.com/en-us/openspecs/windows_protocols/ms-tds/ce3183a6-9d89-47e8-a02f-de5a1a1303de
8/// Column types of the SQL Server source. The `bool` is `true` when the column is nullable.
9#[derive(Copy, Clone, Debug)]
10pub enum MsSQLTypeSystem {
11    /// SQL Server `tinyint` (unsigned 8-bit).
12    Tinyint(bool),
13    /// SQL Server `smallint`.
14    Smallint(bool),
15    /// SQL Server `int`.
16    Int(bool),
17    /// SQL Server `bigint`.
18    Bigint(bool),
19    /// Nullable SQL Server integer of any width, read as `i64`.
20    Intn(bool),
21    /// SQL Server `real` (`float(24)`).
22    Float24(bool),
23    /// SQL Server `float` (`float(53)`).
24    Float53(bool),
25    /// Nullable SQL Server float of either width, read as `f64`.
26    Floatn(bool),
27    /// SQL Server `bit`.
28    Bit(bool),
29    /// SQL Server `nvarchar`.
30    Nvarchar(bool),
31    /// SQL Server `varchar`.
32    Varchar(bool),
33    /// SQL Server `nchar`.
34    Nchar(bool),
35    /// SQL Server `char`.
36    Char(bool),
37    /// SQL Server `ntext`.
38    Ntext(bool),
39    /// SQL Server `text`.
40    Text(bool),
41    /// SQL Server `binary`.
42    Binary(bool),
43    /// SQL Server `varbinary`.
44    Varbinary(bool),
45    /// SQL Server `image`.
46    Image(bool),
47    /// SQL Server `uniqueidentifier`.
48    Uniqueidentifier(bool),
49    /// SQL Server `numeric`.
50    Numeric(bool),
51    /// SQL Server `decimal`.
52    Decimal(bool),
53    /// SQL Server `datetime`.
54    Datetime(bool),
55    /// SQL Server `datetime2`.
56    Datetime2(bool),
57    /// SQL Server `smalldatetime`.
58    Smalldatetime(bool),
59    /// SQL Server `date`.
60    Date(bool),
61    /// SQL Server `time`.
62    Time(bool),
63    /// SQL Server `datetimeoffset`, read as UTC.
64    Datetimeoffset(bool),
65    /// SQL Server `money`, read as `f64`.
66    Money(bool),
67    /// SQL Server `smallmoney`, read as `f32`.
68    SmallMoney(bool),
69}

71impl_typesystem! {
72    system = MsSQLTypeSystem,
73    mappings = {
74        { Tinyint  => u8 }
75        { Smallint => i16 }
76        { Int => i32 }
77        { Bigint => i64 }
78        { Intn => IntN }
79        { Float24 | SmallMoney => f32 }
80        { Float53 | Money => f64 }
81        { Floatn => FloatN }
82        { Bit => bool }
83        { Nvarchar | Varchar | Nchar | Char | Text | Ntext => &'r str }
84        { Binary | Varbinary | Image => &'r [u8] }
85        { Uniqueidentifier => Uuid }
86        { Numeric | Decimal => Decimal }
87        { Datetime | Datetime2 | Smalldatetime => NaiveDateTime }
88        { Date => NaiveDate }
89        { Time => NaiveTime }
90        { Datetimeoffset => DateTime<Utc> }
91    }
92}

95impl<'a> From<&'a ColumnType> for MsSQLTypeSystem {
96    fn from(ty: &'a ColumnType) -> MsSQLTypeSystem {
97        use MsSQLTypeSystem::*;
98
99        match ty {
100            ColumnType::Int1 => Tinyint(false),
101            ColumnType::Int2 => Smallint(false),
102            ColumnType::Int4 => Int(false),
103            ColumnType::Int8 => Bigint(false),
104            ColumnType::Intn => Intn(true),
105            ColumnType::Float4 => Float24(false),
106            ColumnType::Float8 => Float53(false),
107            ColumnType::Floatn => Floatn(true),
108            ColumnType::Bit => Bit(false),
109            ColumnType::Bitn => Bit(true), // nullable int, var-length
110            ColumnType::NVarchar => Nvarchar(true),
111            ColumnType::BigVarChar => Varchar(true),
112            ColumnType::NChar => Nchar(true),
113            ColumnType::BigChar => Char(true),
114            ColumnType::NText => Ntext(true),
115            ColumnType::Text => Text(true),
116            ColumnType::BigBinary => Binary(true),
117            ColumnType::BigVarBin => Varbinary(true),
118            ColumnType::Image => Image(true),
119            ColumnType::Guid => Uniqueidentifier(true),
120            ColumnType::Decimaln => Decimal(true),
121            ColumnType::Numericn => Numeric(true),
122            ColumnType::Datetime => Datetime(false),
123            ColumnType::Datetime2 => Datetime2(true),
124            ColumnType::Datetimen => Datetime(true),
125            ColumnType::Datetime4 => Datetime(false),
126            ColumnType::Daten => Date(true),
127            ColumnType::Timen => Time(true),
128            ColumnType::DatetimeOffsetn => Datetimeoffset(true),
129            ColumnType::Money => Money(true),
130            ColumnType::Money4 => SmallMoney(true),
131            _ => unimplemented!("{}", format!("{:?}", ty)),
132        }
133    }
134}

136/// `mssql-tds`'s wire-level metadata is coarser than Tiberius's: nullable
137/// int/float columns are all reported as a single `IntN`/`FltN` variant
138/// regardless of byte width (the actual width is only known once a row is
139/// decoded into a [`mssql_tds::datatypes::column_values::ColumnValues`]).
140/// This mirrors how `MsSQLTypeSystem::Intn`/`Floatn` already behave on the
141/// Tiberius path today, so no new type-system variants are needed.
142#[cfg(feature = "src_mssql_tds")]
143impl<'a> From<&'a mssql_tds::query::metadata::ColumnMetadata> for MsSQLTypeSystem {
144    fn from(meta: &'a mssql_tds::query::metadata::ColumnMetadata) -> MsSQLTypeSystem {
145        use mssql_tds::datatypes::sqldatatypes::TdsDataType;
146        use MsSQLTypeSystem::*;
147
148        let nullable = meta.is_nullable();
149        match meta.data_type {
150            TdsDataType::Int1 => Tinyint(nullable),
151            TdsDataType::Int2 => Smallint(nullable),
152            TdsDataType::Int4 => Int(nullable),
153            TdsDataType::Int8 => Bigint(nullable),
154            TdsDataType::IntN => Intn(true),
155            TdsDataType::Flt4 => Float24(nullable),
156            TdsDataType::Flt8 => Float53(nullable),
157            TdsDataType::FltN => Floatn(true),
158            TdsDataType::Bit | TdsDataType::BitN => Bit(nullable),
159            TdsDataType::NVarChar => Nvarchar(true),
160            TdsDataType::BigVarChar => Varchar(true),
161            TdsDataType::NChar => Nchar(true),
162            TdsDataType::BigChar => Char(true),
163            TdsDataType::NText => Ntext(true),
164            TdsDataType::Text => Text(true),
165            TdsDataType::BigBinary => Binary(true),
166            TdsDataType::BigVarBinary => Varbinary(true),
167            TdsDataType::Image => Image(true),
168            TdsDataType::Guid => Uniqueidentifier(true),
169            TdsDataType::DecimalN | TdsDataType::Decimal => Decimal(true),
170            TdsDataType::NumericN | TdsDataType::Numeric => Numeric(true),
171            TdsDataType::DateTime | TdsDataType::DateTim4 => Datetime(nullable),
172            TdsDataType::DateTimeN => Datetime(true),
173            TdsDataType::DateTime2N => Datetime2(true),
174            TdsDataType::DateN => Date(true),
175            TdsDataType::TimeN => Time(true),
176            TdsDataType::DateTimeOffsetN => Datetimeoffset(true),
177            // Only a fixed-width `Money4` wire type (non-nullable smallmoney)
178            // is treated as SmallMoney/f32. A nullable smallmoney column is
179            // sent as the generic `MoneyN` wire type with a per-row length
180            // (4 or 8 bytes) that isn't known statically from metadata alone
181            // - matching Tiberius, which always maps its nullable `MONEYN`
182            // wire type to F64 regardless of the per-row byte length (see
183            // tiberius's `money::decode`), so parity requires treating
184            // `MoneyN` as `Money` (f64) here too, never `SmallMoney`.
185            TdsDataType::Money => Money(nullable),
186            TdsDataType::MoneyN => Money(true),
187            TdsDataType::Money4 => SmallMoney(nullable),
188            ref ty => unimplemented!("{}", format!("{:?}", ty)),
189        }
190    }
191}

193/// Value of a nullable integer column of any width, widened to `i64`.
194pub struct IntN(pub i64);
195#[cfg(feature = "src_mssql_tiberius")]
196impl<'a> FromSql<'a> for IntN {
197    fn from_sql(value: &'a ColumnData<'static>) -> Result<Option<Self>, tiberius::error::Error> {
198        match value {
199            ColumnData::U8(None)
200            | ColumnData::I16(None)
201            | ColumnData::I32(None)
202            | ColumnData::I64(None) => Ok(None),
203            ColumnData::U8(Some(d)) => Ok(Some(IntN(*d as i64))),
204            ColumnData::I16(Some(d)) => Ok(Some(IntN(*d as i64))),
205            ColumnData::I32(Some(d)) => Ok(Some(IntN(*d as i64))),
206            ColumnData::I64(Some(d)) => Ok(Some(IntN(*d))),
207            v => Err(tiberius::error::Error::Conversion(
208                format!("cannot interpret {:?} as a intn value", v).into(),
209            )),
210        }
211    }
212}

214/// Value of a nullable float column of either width, widened to `f64`.
215pub struct FloatN(pub f64);
216#[cfg(feature = "src_mssql_tiberius")]
217impl<'a> FromSql<'a> for FloatN {
218    fn from_sql(value: &'a ColumnData<'static>) -> Result<Option<Self>, tiberius::error::Error> {
219        match value {
220            ColumnData::F32(None) | ColumnData::F64(None) => Ok(None),
221            ColumnData::F32(Some(d)) => Ok(Some(FloatN(*d as f64))),
222            ColumnData::F64(Some(d)) => Ok(Some(FloatN(*d))),
223            v => Err(tiberius::error::Error::Conversion(
224                format!("cannot interpret {:?} as a floatn value", v).into(),
225            )),
226        }
227    }
228}

241            (ColumnType::Int8, "Bigint(false)"),
242            (ColumnType::Intn, "Intn"),
243            (ColumnType::Float4, "Float24(false)"),
244            (ColumnType::Float8, "Float53(false)"),
245            (ColumnType::Floatn, "Floatn"),
246            (ColumnType::Bit, "Bit(false)"),
247            (ColumnType::Bitn, "Bit"),
248            (ColumnType::NVarchar, "Nvarchar"),
249            (ColumnType::BigVarChar, "Varchar"),
250            (ColumnType::NChar, "Nchar"),
251            (ColumnType::BigChar, "Char"),
252            (ColumnType::NText, "Ntext"),
253            (ColumnType::Text, "Text"),
254            (ColumnType::BigBinary, "Binary"),
255            (ColumnType::BigVarBin, "Varbinary"),
256            (ColumnType::Image, "Image"),
257            (ColumnType::Guid, "Uniqueidentifier"),
258            (ColumnType::Decimaln, "Decimal"),
259            (ColumnType::Numericn, "Numeric"),
260            (ColumnType::Datetime, "Datetime(false)"),
261            (ColumnType::Datetime2, "Datetime2"),
262            (ColumnType::Datetimen, "Datetime"),

264            (ColumnType::Daten, "Date"),
265            (ColumnType::Timen, "Time"),
266            (ColumnType::DatetimeOffsetn, "Datetimeoffset"),
267            (ColumnType::Money, "Money"),
268            (ColumnType::Money4, "SmallMoney"),
269        ];
270
271        for (ty, expected) in cases {
272            assert_eq!(
273                format!("{:?}", MsSQLTypeSystem::from(&ty)),
274                if expected.contains('(') {
275                    expected.to_owned()
276                } else {
277                    format!("{expected}(true)")
278                }
279            );
280        }
281        assert!(matches!(
282            MsSQLTypeSystem::from(&ColumnType::Int1),
283            MsSQLTypeSystem::Tinyint(false)
284        ));
285        assert!(matches!(
286            MsSQLTypeSystem::from(&ColumnType::Datetime),
287            MsSQLTypeSystem::Datetime(false)
288        ));
289        assert!(matches!(
290            MsSQLTypeSystem::from(&ColumnType::Datetime4),
291            MsSQLTypeSystem::Datetime(false)
292        ));
293    }

295    #[test]
296    fn parses_nullable_integer_and_float_values() {
297        let integer_values = [
298            ColumnData::U8(Some(1)),
299            ColumnData::I16(Some(2)),
300            ColumnData::I32(Some(3)),
301            ColumnData::I64(Some(4)),
302        ];
303        for (value, expected) in integer_values.iter().zip(1..=4) {
304            assert_eq!(IntN::from_sql(value).unwrap().unwrap().0, expected);
305        }
306        for value in [
307            ColumnData::U8(None),
308            ColumnData::I16(None),
309            ColumnData::I32(None),
310            ColumnData::I64(None),
311        ] {
312            assert!(IntN::from_sql(&value).unwrap().is_none());
313        }
314        assert!(IntN::from_sql(&ColumnData::Bit(Some(true))).is_err());
315
316        let float_values = [ColumnData::F32(Some(1.5)), ColumnData::F64(Some(2.5))];
317        assert_eq!(FloatN::from_sql(&float_values[0]).unwrap().unwrap().0, 1.5);
318        assert_eq!(FloatN::from_sql(&float_values[1]).unwrap().unwrap().0, 2.5);
319        assert!(FloatN::from_sql(&ColumnData::F32(None)).unwrap().is_none());
320        assert!(FloatN::from_sql(&ColumnData::F64(None)).unwrap().is_none());
321        assert!(FloatN::from_sql(&ColumnData::Bit(Some(true))).is_err());
322    }
