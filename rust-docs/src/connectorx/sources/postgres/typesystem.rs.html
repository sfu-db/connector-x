11/// Column types of the Postgres source. The `bool` is `true` when the column is nullable.

23    Int2(bool),
24    /// Postgres `int4` (32-bit integer).
25    Int4(bool),
26    /// Postgres `int8` (64-bit integer).
27    Int8(bool),
28    /// Postgres `oid` (unsigned 32-bit integer).
29    UInt4(bool),
30    /// Postgres `float4[]`.
31    Float4Array(bool),
32    /// Postgres `float8[]`.
33    Float8Array(bool),
34    /// Postgres `numeric[]`.
35    NumericArray(bool),
36    /// Postgres `bool[]`.
37    BoolArray(bool),

39    Int2Array(bool),

45    VarcharArray(bool),
46    /// Postgres `text[]`.

94impl_typesystem! {
95    system = PostgresTypeSystem,
96    mappings = {
97        { Int2 => i16 }
98        { Int4 => i32 }
99        { Int8 => i64 }
100        { UInt4 => u32 }
101        { Float4 => f32 }
102        { Float8 => f64 }
103        { Numeric => Decimal }
104        { BoolArray => Vec<Option<bool>> }
105        { Int2Array => Vec<Option<i16>> }
106        { Int4Array => Vec<Option<i32>> }
107        { Int8Array => Vec<Option<i64>> }
108        { Float4Array => Vec<Option<f32>> }
109        { Float8Array => Vec<Option<f64>> }
110        { NumericArray => Vec<Option<Decimal>> }
111        { VarcharArray | TextArray => Vec<Option<String>>}
112        { Bool => bool }
113        { Char => i8 }
114        { Text | BpChar | VarChar | Enum | Name | Range | TsVector => &'r str }
115        { ByteA => Vec<u8> }
116        { Time => NaiveTime }
117        { Timestamp => NaiveDateTime }
118        { TimestampTz => DateTime<Utc> }
119        { Date => NaiveDate }
120        { UUID => Uuid }
121        { JSON | JSONB => Value }
122        { HSTORE => HashMap<String, Option<String>> }
123        { Inet => IpInet }
124        { Vector => Vector }
125        { HalfVec => HalfVector }
126        { Bit => Bit }
127        { SparseVec => SparseVector }
128    }
129}

131impl<'a> From<&'a Type> for PostgresTypeSystem {
132    fn from(ty: &'a Type) -> PostgresTypeSystem {
133        use PostgresTypeSystem::*;
134        match ty.name() {
135            "int2" => Int2(true),
136            "int4" => Int4(true),
137            "int8" => Int8(true),
138            "oid" => UInt4(true),
139            "float4" => Float4(true),
140            "float8" => Float8(true),
141            "numeric" => Numeric(true),
142            "_bool" => BoolArray(true),
143            "_int2" => Int2Array(true),
144            "_int4" => Int4Array(true),
145            "_int8" => Int8Array(true),
146            "_float4" => Float4Array(true),
147            "_float8" => Float8Array(true),
148            "_numeric" => NumericArray(true),
149            "_varchar" => VarcharArray(true),
150            "_text" => TextArray(true),
151            "bool" => Bool(true),
152            "char" => Char(true),
153            "text" | "citext" | "ltree" | "lquery" | "ltxtquery" | "name" => Text(true),
154            "bpchar" => BpChar(true),
155            "varchar" => VarChar(true),
156            "bytea" => ByteA(true),
157            "time" => Time(true),
158            "timestamp" => Timestamp(true),
159            "timestamptz" => TimestampTz(true),
160            "date" => Date(true),
161            "uuid" => UUID(true),
162            "json" => JSON(true),
163            "jsonb" => JSONB(true),
164            "hstore" => HSTORE(true),
165            "inet" => Inet(true),
166            "tsvector" => TsVector(true),
167            "int4range" | "int8range" | "numrange" | "tsrange" | "tstzrange" | "daterange" => {
168                Range(true)
169            }
170            "vector" => Vector(true),
171            "halfvec" => HalfVec(true),
172            "bit" => Bit(true),
173            "sparsevec" => SparseVec(true),
174            _ => match ty.kind() {
175                postgres::types::Kind::Enum(_) => Enum(true),
176                _ => unimplemented!("{}", ty.name()),
177            },
178        }
179    }
180}

182/// A Postgres column type paired with the ConnectorX type chosen for it; converts back to the `Type` to request on the wire.
183pub struct PostgresTypePairs<'a>(pub &'a Type, pub &'a PostgresTypeSystem);
184
185// Link (postgres::Type, connectorx::PostgresTypes) back to the one defined by the postgres crate.
186impl From<PostgresTypePairs<'_>> for Type {
187    fn from(ty: PostgresTypePairs) -> Type {
188        use PostgresTypeSystem::*;
189        match ty.1 {
190            Enum(_) => Type::TEXT,
191            HSTORE(_) => Type::TEXT, // hstore is not supported in binary protocol (since no corresponding inner TYPE)
192            Range(_) | TsVector(_) => Type::TEXT,
193            _ => ty.0.clone(),
194        }
195    }
196}
