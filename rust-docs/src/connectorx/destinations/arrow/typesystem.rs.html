1use chrono::{DateTime, NaiveDate, NaiveDateTime, NaiveTime, Utc};

4/// UTC timestamp that is written with microsecond precision.
5#[derive(Debug, Clone, Copy)]
6pub struct DateTimeWrapperMicro(pub DateTime<Utc>);
7
8/// Time of day that is written with microsecond precision.
9#[derive(Debug, Clone, Copy)]
10pub struct NaiveTimeWrapperMicro(pub NaiveTime);
11
12/// Date and time without time zone that is written with microsecond precision.
13#[derive(Debug, Clone, Copy)]

16/// Column types of the Arrow destination. The `bool` is `true` when the column is nullable.
17#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash, PartialOrd, Ord)]
18pub enum ArrowTypeSystem {
19    /// Arrow `Int8`.
20    Int8(bool),
21    /// Arrow `Int16`.
22    Int16(bool),
23    /// Arrow `Int32`.
24    Int32(bool),
25    /// Arrow `Int64`.
26    Int64(bool),
27    /// Arrow `UInt8`.
28    UInt8(bool),
29    /// Arrow `UInt16`.
30    UInt16(bool),
31    /// Arrow `UInt32`.
32    UInt32(bool),
33    /// Arrow `UInt64`.
34    UInt64(bool),
35    /// Arrow `Float32`.
36    Float32(bool),
37    /// Arrow `Float64`.
38    Float64(bool),
39    /// Arrow `Decimal128(38, 10)`.
40    Decimal(bool),
41    /// Arrow `Boolean`.
42    Boolean(bool),
43    /// String, written as Arrow `Utf8`.
44    LargeUtf8(bool),
45    /// Arrow `LargeBinary`.
46    LargeBinary(bool),
47    /// Arrow `Date32` (days since the epoch).
48    Date32(bool),
49    /// Date and time without time zone, written as Arrow `Timestamp(Nanosecond, None)`.
50    Date64(bool),
51    /// Date and time without time zone, written as Arrow `Timestamp(Microsecond, None)`.
52    Date64Micro(bool),
53    /// Arrow `Time64(Nanosecond)`.
54    Time64(bool),
55    /// Arrow `Time64(Microsecond)`.
56    Time64Micro(bool),
57    /// UTC timestamp, written as Arrow `Timestamp(Nanosecond, None)`.
58    DateTimeTz(bool),
59    /// UTC timestamp, written as Arrow `Timestamp(Microsecond, "+00:00")`.
60    DateTimeTzMicro(bool),
61    /// Arrow `LargeList<Boolean>`.
62    BoolArray(bool),
63    /// Arrow `LargeList<Utf8>`.
64    Utf8Array(bool),
65    /// Arrow `LargeList<Int16>`.
66    Int16Array(bool),
67    /// Arrow `LargeList<Int32>`.
68    Int32Array(bool),
69    /// Arrow `LargeList<Int64>`.
70    Int64Array(bool),
71    /// Arrow `LargeList<UInt16>`.
72    UInt16Array(bool),
73    /// Arrow `LargeList<UInt32>`.
74    UInt32Array(bool),
75    /// Arrow `LargeList<UInt64>`.
76    UInt64Array(bool),
77    /// Arrow `LargeList<Float32>`.
78    Float32Array(bool),
79    /// Arrow `LargeList<Float64>`.
80    Float64Array(bool),
81    /// Arrow `LargeList<Decimal128(38, 10)>`.
82    DecimalArray(bool),
83}

85impl_typesystem! {
86    system = ArrowTypeSystem,
87    mappings = {
88        { Int8            => i8                        }
89        { Int16           => i16                       }
90        { Int32           => i32                       }
91        { Int64           => i64                       }
92        { UInt8           => u8                        }
93        { UInt16          => u16                       }
94        { UInt32          => u32                       }
95        { UInt64          => u64                       }
96        { Float32         => f32                       }
97        { Float64         => f64                       }
98        { Decimal         => Decimal                   }
99        { Boolean         => bool                      }
100        { LargeUtf8       => String                    }
101        { LargeBinary     => Vec<u8>                   }
102        { Date32          => NaiveDate                 }
103        { Date64          => NaiveDateTime             }
104        { Date64Micro     => NaiveDateTimeWrapperMicro }
105        { Time64          => NaiveTime                 }
106        { Time64Micro     => NaiveTimeWrapperMicro     }
107        { DateTimeTz      => DateTime<Utc>             }
108        { DateTimeTzMicro => DateTimeWrapperMicro      }
109        { BoolArray       => Vec<Option<bool>>         }
110        { Utf8Array       => Vec<Option<String>>       }
111        { Int16Array      => Vec<Option<i16>>          }
112        { Int32Array      => Vec<Option<i32>>          }
113        { Int64Array      => Vec<Option<i64>>          }
114        { UInt16Array     => Vec<Option<u16>>          }
115        { UInt32Array     => Vec<Option<u32>>          }
116        { UInt64Array     => Vec<Option<u64>>          }
117        { Float32Array    => Vec<Option<f32>>          }
118        { Float64Array    => Vec<Option<f64>>          }
119        { DecimalArray    => Vec<Option<Decimal>>      }
120    }
