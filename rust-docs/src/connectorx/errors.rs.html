172/// Errors that can be raised from this library.
173#[derive(Error, Debug)]
174pub enum ConnectorXError {
175    /// The required type does not same as the schema defined.
176    #[error("Data type unexpected: {0:?} expected, {1} found.")]
177    TypeCheckFailed(String, &'static str),
178
179    #[error("Data order not supported {0:?}.")]
180    UnsupportedDataOrder(DataOrder),
181
182    #[error("Cannot resolve data order: got {0:?} from source, {1:?} from destination.")]
183    CannotResolveDataOrder(Vec<DataOrder>, Vec<DataOrder>),
184
185    #[error("Cannot produce a {0}, context: {1}.")]
186    CannotProduce(&'static str, ProduceContext),
187
188    #[error("No conversion rule from {0} to {1}.")]
189    NoConversionRule(String, String),
190
191    #[error("Only support single query with SELECT statement, got {0}.")]
192    SqlQueryNotSupported(String),
193
194    #[error("Cannot get total number of rows in advance.")]
195    CountError(),
196
197    #[error(transparent)]
198    SQLParserError(#[from] sqlparser::parser::ParserError),
199
200    #[error(transparent)]
201    StdIOError(#[from] std::io::Error),
202
203    #[error(transparent)]
204    StdVarError(#[from] std::env::VarError),
205
206    #[error(transparent)]
207    Other(#[from] anyhow::Error),
208}

210impl ConnectorXError {
211    pub fn cannot_produce<T>(context: Option<String>) -> Self {
212        ConnectorXError::CannotProduce(type_name::<T>(), context.into())
213    }
214}
215
216#[derive(Debug)]
217pub enum ProduceContext {
218    NoContext,
219    Context(String),
220}
221
222impl From<Option<String>> for ProduceContext {
223    fn from(val: Option<String>) -> Self {
224        match val {
225            Some(c) => ProduceContext::Context(c),
226            None => ProduceContext::NoContext,
227        }
228    }
229}
230
231impl fmt::Display for ProduceContext {
232    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
233        match self {
234            ProduceContext::NoContext => write!(f, "No Context"),
235            ProduceContext::Context(s) => write!(f, "{}", s),
236        }
237    }
238}
